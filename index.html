My research investigates how artificial intelligence interacts with complex social and infrastructural systems. I combine machine learning, complex systems science, and philosophy of technology to study how algorithmic decision systems shape collective behavior, governance, and large-scale coordination. A central goal of my work is developing AI systems that remain robust, transparent, and socially aligned when deployed in real-world socio-technical environments.

function toggleTheme(){document.body.classList.toggle("dark")}


// Load Scholar stats from local JSON (i10-index included)

async function loadScholarStats(){

 try{

  const res = await fetch('scholar-stats.json');

  const data = await res.json();

  document.getElementById('scholar-stats').innerHTML = `

    <div style="margin-top:20px;font-size:15px">

      <strong>${data.citations}</strong> citations · 

      <strong>${data.hindex}</strong> h-index · 

      <strong>${data.i10index}</strong> i10-index

    </div>

  `;

 }catch(e){console.log('Scholar stats not loaded — run GitHub Action')}

}

loadScholarStats();


// Load publications with abstract, authors, venue, sorted by year descending

async function loadPublications(){

  try{

    const res = await fetch("publications.json");

    const papers = await res.json();

    const container = document.getElementById("pub-list");

    container.innerHTML = "";

    papers.forEach(p => {

      const div = document.createElement("div");

      div.className = "paper";

      div.innerHTML = `

        <div class="paper-title">${p.title}</div>

        <div class="abstract">

          <strong>Authors:</strong> ${p.authors}<br>

          <strong>Venue:</strong> ${p.venue} (${p.year})<br>

          ${p.abstract ? "<strong>Abstract:</strong> " + p.abstract : ""}

        </div>

      `;

      container.appendChild(div);

    });

  } catch(e){ console.log("Failed to load publications:", e); }

}

loadPublications();
